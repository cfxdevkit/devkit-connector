-- =============================================
-- CONFLUX DUAL WALLET SYSTEM DATABASE SCHEMA
-- =============================================

-- =============================================
-- 1. USER MANAGEMENT & AUTHENTICATION
-- =============================================

-- Stores user accounts and authentication data
CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    username VARCHAR(50) UNIQUE,
    password_hash VARCHAR(255), -- For traditional auth (optional)
    auth_provider VARCHAR(50), -- 'email', 'google', 'github', etc.
    auth_provider_id VARCHAR(255), -- External provider user ID
    email_verified BOOLEAN DEFAULT FALSE,
    two_factor_enabled BOOLEAN DEFAULT FALSE,
    two_factor_secret VARCHAR(255), -- TOTP secret
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    last_login TIMESTAMP,
    status VARCHAR(20) DEFAULT 'active', -- 'active', 'suspended', 'deleted'
    metadata JSONB -- Additional user preferences, settings
);

-- User sessions for authentication (JWT tracking)
CREATE TABLE user_sessions (
    session_id VARCHAR(128) PRIMARY KEY,
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    device_info JSONB, -- Browser, OS, IP address
    created_at TIMESTAMP DEFAULT NOW(),
    expires_at TIMESTAMP NOT NULL,
    last_activity TIMESTAMP DEFAULT NOW(),
    is_active BOOLEAN DEFAULT TRUE
);

-- =============================================
-- 2. OPTION A: SERVER-MANAGED WALLETS
-- =============================================

-- Stores encrypted mnemonics for server-managed wallets
CREATE TABLE server_wallets (
    wallet_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    encrypted_mnemonic TEXT NOT NULL, -- AES-256 encrypted mnemonic
    encryption_iv VARCHAR(32) NOT NULL, -- Initialization vector for encryption
    encryption_method VARCHAR(50) DEFAULT 'aes-256-cbc',
    espace_address VARCHAR(42) NOT NULL, -- 0x... format
    core_address VARCHAR(64) NOT NULL, -- cfx:... format
    derivation_path VARCHAR(100) DEFAULT "m/44'/60'/0'/0/0",
    network_id INTEGER DEFAULT 1029, -- Conflux network ID
    wallet_version INTEGER DEFAULT 1, -- For future wallet upgrades
    backup_status VARCHAR(20) DEFAULT 'pending', -- 'pending', 'completed', 'failed'
    backup_location TEXT, -- Where backup is stored (encrypted)
    created_at TIMESTAMP DEFAULT NOW(),
    last_accessed TIMESTAMP,
    access_count INTEGER DEFAULT 0,
    status VARCHAR(20) DEFAULT 'active' -- 'active', 'locked', 'archived'
);

-- Track server wallet usage and access patterns
CREATE TABLE server_wallet_access_log (
    log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    wallet_id UUID NOT NULL REFERENCES server_wallets(wallet_id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    access_type VARCHAR(50) NOT NULL, -- 'load', 'sign_transaction', 'sign_message'
    ip_address INET,
    user_agent TEXT,
    success BOOLEAN NOT NULL,
    error_message TEXT,
    timestamp TIMESTAMP DEFAULT NOW(),
    request_data JSONB -- Sanitized request parameters
);

-- =============================================
-- 3. OPTION B: USER-DELEGATED WALLETS
-- =============================================

-- Main delegation sessions table
CREATE TABLE delegation_sessions (
    session_id VARCHAR(64) PRIMARY KEY,
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    user_wallet_address VARCHAR(42) NOT NULL, -- User's original wallet address
    espace_address VARCHAR(42) NOT NULL, -- Derived eSpace address
    core_address VARCHAR(64) NOT NULL, -- Derived Core address
    signature_proof TEXT NOT NULL, -- Signature proving wallet ownership
    delegation_message TEXT NOT NULL, -- Original message that was signed
    
    -- Session timing
    created_at TIMESTAMP DEFAULT NOW(),
    expires_at TIMESTAMP NOT NULL,
    last_activity TIMESTAMP DEFAULT NOW(),
    auto_renew BOOLEAN DEFAULT FALSE,
    max_renewals INTEGER DEFAULT 0,
    current_renewals INTEGER DEFAULT 0,
    
    -- Session status
    status VARCHAR(20) DEFAULT 'active', -- 'active', 'expired', 'revoked', 'suspended'
    revoked_at TIMESTAMP,
    revoked_reason TEXT,
    
    -- Configuration
    config JSONB NOT NULL, -- Full delegation configuration
    config_hash VARCHAR(64), -- Hash of config for integrity checking
    
    -- Security
    ip_address INET,
    user_agent TEXT,
    device_fingerprint VARCHAR(255)
);

-- Track all delegation session changes and events
CREATE TABLE delegation_session_events (
    event_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id VARCHAR(64) NOT NULL REFERENCES delegation_sessions(session_id) ON DELETE CASCADE,
    event_type VARCHAR(50) NOT NULL, 
    /* Event types:
       - 'session_created', 'session_renewed', 'session_expired', 'session_revoked'
       - 'config_updated', 'permissions_changed', 'limits_modified'
       - 'security_alert', 'suspicious_activity', 'rate_limit_hit'
    */
    event_data JSONB NOT NULL, -- Detailed event information
    triggered_by VARCHAR(50), -- 'user', 'system', 'auto_renewal', 'security_check'
    ip_address INET,
    timestamp TIMESTAMP DEFAULT NOW()
);

-- =============================================
-- 4. OPERATIONS & TRANSACTIONS (BOTH OPTIONS)
-- =============================================

-- All wallet operations regardless of wallet type
CREATE TABLE wallet_operations (
    operation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- Identification
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    wallet_id UUID REFERENCES server_wallets(wallet_id) ON DELETE SET NULL, -- For Option A
    session_id VARCHAR(64) REFERENCES delegation_sessions(session_id) ON DELETE SET NULL, -- For Option B
    wallet_type VARCHAR(20) NOT NULL, -- 'server_managed', 'user_delegated'
    
    -- Operation details
    operation_type VARCHAR(50) NOT NULL,
    /* Operation types:
       - 'sign_transaction', 'sign_message', 'get_balance', 'estimate_gas'
       - 'deploy_contract', 'interact_contract', 'transfer_tokens'
       - 'swap_tokens', 'stake_tokens', 'bridge_tokens'
    */
    chain_type VARCHAR(10) NOT NULL, -- 'eSpace', 'core'
    
    -- Status tracking
    status VARCHAR(20) NOT NULL DEFAULT 'pending',
    /* Status values:
       - 'pending', 'approved', 'rejected', 'executing', 'completed', 'failed'
       - 'requires_approval', 'auto_approved', 'user_approved'
    */
    
    -- Operation data
    operation_payload JSONB NOT NULL, -- Input parameters (transaction details, etc.)
    operation_result JSONB, -- Output results (transaction hash, signature, etc.)
    error_details JSONB, -- Error information if failed
    
    -- Approval workflow
    requires_user_approval BOOLEAN DEFAULT FALSE,
    auto_approval_rule_id VARCHAR(64), -- Which rule auto-approved this
    user_approved_at TIMESTAMP,
    approved_by UUID REFERENCES users(user_id),
    approval_signature TEXT, -- User signature for approval
    
    -- Execution details
    gas_estimate BIGINT,
    gas_used BIGINT,
    gas_price BIGINT,
    transaction_hash VARCHAR(66), -- Blockchain transaction hash
    block_number BIGINT,
    confirmation_count INTEGER DEFAULT 0,
    
    -- Timing
    created_at TIMESTAMP DEFAULT NOW(),
    approved_at TIMESTAMP,
    executed_at TIMESTAMP,
    completed_at TIMESTAMP,
    
    -- Security & audit
    ip_address INET,
    user_agent TEXT,
    risk_score NUMERIC(3,2), -- 0.00 to 1.00 risk assessment
    fraud_check_result JSONB -- Anti-fraud analysis results
);

-- Track operation state changes for audit trail
CREATE TABLE operation_state_changes (
    change_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    operation_id UUID NOT NULL REFERENCES wallet_operations(operation_id) ON DELETE CASCADE,
    from_status VARCHAR(20),
    to_status VARCHAR(20) NOT NULL,
    reason TEXT,
    changed_by VARCHAR(50), -- 'system', 'user', 'auto_approval', 'admin'
    changed_at TIMESTAMP DEFAULT NOW(),
    additional_data JSONB
);

-- =============================================
-- 5. AUTO-APPROVAL RULES & CONFIGURATION
-- =============================================

-- Store auto-approval rules for delegation sessions
CREATE TABLE auto_approval_rules (
    rule_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id VARCHAR(64) NOT NULL REFERENCES delegation_sessions(session_id) ON DELETE CASCADE,
    rule_name VARCHAR(100),
    rule_description TEXT,
    rule_type VARCHAR(50) NOT NULL, -- 'transaction_value', 'contract_address', etc.
    condition_type VARCHAR(50) NOT NULL, -- 'less_than', 'whitelist', etc.
    condition_value JSONB NOT NULL, -- The actual condition parameters
    auto_approve BOOLEAN NOT NULL,
    priority INTEGER DEFAULT 100, -- Lower number = higher priority
    chain_specific VARCHAR(10), -- 'eSpace', 'core', or NULL for both
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    usage_count INTEGER DEFAULT 0, -- How many times this rule has been applied
    last_used TIMESTAMP
);

-- Track when auto-approval rules are triggered
CREATE TABLE rule_applications (
    application_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    rule_id UUID NOT NULL REFERENCES auto_approval_rules(rule_id) ON DELETE CASCADE,
    operation_id UUID NOT NULL REFERENCES wallet_operations(operation_id) ON DELETE CASCADE,
    session_id VARCHAR(64) NOT NULL,
    rule_matched BOOLEAN NOT NULL,
    rule_result VARCHAR(20) NOT NULL, -- 'approved', 'rejected', 'skipped'
    match_details JSONB, -- Why the rule matched/didn't match
    applied_at TIMESTAMP DEFAULT NOW()
);

-- =============================================
-- 6. SECURITY & MONITORING
-- =============================================

-- Security events and alerts
CREATE TABLE security_events (
    event_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    session_id VARCHAR(64) REFERENCES delegation_sessions(session_id) ON DELETE SET NULL,
    wallet_id UUID REFERENCES server_wallets(wallet_id) ON DELETE SET NULL,
    
    event_type VARCHAR(50) NOT NULL,
    /* Security event types:
       - 'suspicious_login', 'rate_limit_exceeded', 'unusual_transaction_pattern'
       - 'large_transaction_attempted', 'unknown_contract_interaction'
       - 'geographic_anomaly', 'device_change', 'permission_escalation_attempt'
    */
    
    severity VARCHAR(20) NOT NULL, -- 'low', 'medium', 'high', 'critical'
    event_data JSONB NOT NULL,
    automatic_action_taken VARCHAR(100), -- 'session_suspended', 'approval_required', etc.
    resolution_status VARCHAR(20) DEFAULT 'open', -- 'open', 'investigating', 'resolved', 'false_positive'
    resolved_at TIMESTAMP,
    resolved_by UUID REFERENCES users(user_id),
    
    created_at TIMESTAMP DEFAULT NOW(),
    ip_address INET,
    user_agent TEXT,
    location_data JSONB -- Geographic information if available
);

-- Rate limiting and usage tracking
CREATE TABLE rate_limits (
    limit_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    session_id VARCHAR(64) REFERENCES delegation_sessions(session_id) ON DELETE CASCADE,
    limit_type VARCHAR(50) NOT NULL, -- 'operations_per_hour', 'transactions_per_day', 'value_per_day'
    limit_value BIGINT NOT NULL,
    current_usage BIGINT DEFAULT 0,
    reset_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT NOW()
);

-- =============================================
-- 7. ANALYTICS & REPORTING
-- =============================================

-- Daily aggregated statistics
CREATE TABLE daily_statistics (
    stat_date DATE NOT NULL,
    metric_type VARCHAR(50) NOT NULL, -- 'transactions', 'volume', 'users', 'sessions'
    metric_category VARCHAR(50), -- 'espace', 'core', 'server_managed', 'delegated'
    metric_value BIGINT NOT NULL,
    additional_data JSONB,
    PRIMARY KEY (stat_date, metric_type, metric_category)
);

-- User behavior analytics
CREATE TABLE user_analytics (
    analytics_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    analysis_date DATE NOT NULL,
    
    -- Usage patterns
    total_operations INTEGER DEFAULT 0,
    successful_operations INTEGER DEFAULT 0,
    failed_operations INTEGER DEFAULT 0,
    average_transaction_value NUMERIC(78,18),
    preferred_chain VARCHAR(10), -- 'eSpace' or 'core'
    preferred_wallet_type VARCHAR(20), -- 'server_managed' or 'user_delegated'
    
    -- Timing patterns
    most_active_hour INTEGER, -- 0-23
    most_active_day INTEGER, -- 1-7 (Monday-Sunday)
    session_duration_avg INTERVAL,
    
    -- Risk assessment
    risk_score NUMERIC(3,2),
    security_incidents INTEGER DEFAULT 0,
    
    -- Additional metrics
    metrics JSONB,
    
    PRIMARY KEY (user_id, analysis_date)
);

-- =============================================
-- 8. NOTIFICATIONS & WEBHOOKS
-- =============================================

-- Notification preferences and delivery tracking
CREATE TABLE notifications (
    notification_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    session_id VARCHAR(64) REFERENCES delegation_sessions(session_id) ON DELETE SET NULL,
    operation_id UUID REFERENCES wallet_operations(operation_id) ON DELETE SET NULL,
    
    notification_type VARCHAR(50) NOT NULL, -- 'email', 'webhook', 'push', 'sms'
    event_type VARCHAR(50) NOT NULL, -- 'transaction_completed', 'approval_required', etc.
    
    recipient VARCHAR(255) NOT NULL, -- Email, webhook URL, phone number, etc.
    subject VARCHAR(255),
    content TEXT,
    
    status VARCHAR(20) DEFAULT 'pending', -- 'pending', 'sent', 'delivered', 'failed'
    attempts INTEGER DEFAULT 0,
    max_attempts INTEGER DEFAULT 3,
    
    scheduled_at TIMESTAMP DEFAULT NOW(),
    sent_at TIMESTAMP,
    delivered_at TIMESTAMP,
    failed_at TIMESTAMP,
    error_message TEXT,
    
    response_data JSONB -- Webhook response, email delivery status, etc.
);

-- Webhook endpoints for external notifications
CREATE TABLE webhook_endpoints (
    endpoint_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    session_id VARCHAR(64) REFERENCES delegation_sessions(session_id) ON DELETE CASCADE,
    
    url TEXT NOT NULL,
    secret_key VARCHAR(255), -- For webhook signature verification
    event_types TEXT[] NOT NULL, -- Which events to send to this webhook
    is_active BOOLEAN DEFAULT TRUE,
    
    created_at TIMESTAMP DEFAULT NOW(),
    last_used TIMESTAMP,
    success_count INTEGER DEFAULT 0,
    failure_count INTEGER DEFAULT 0
);

-- =============================================
-- 9. BACKUP & RECOVERY
-- =============================================

-- Encrypted backups of critical data
CREATE TABLE encrypted_backups (
    backup_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    backup_type VARCHAR(50) NOT NULL, -- 'wallet_mnemonic', 'delegation_keys', 'user_data'
    
    encrypted_data TEXT NOT NULL, -- The actual encrypted backup
    encryption_method VARCHAR(50) NOT NULL,
    encryption_key_id VARCHAR(100), -- Reference to key management system
    checksum VARCHAR(64), -- Data integrity verification
    
    created_at TIMESTAMP DEFAULT NOW(),
    expires_at TIMESTAMP,
    download_count INTEGER DEFAULT 0,
    last_accessed TIMESTAMP
);

-- Recovery codes for emergency access
CREATE TABLE recovery_codes (
    code_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    code_hash VARCHAR(255) NOT NULL, -- Hashed recovery code
    code_type VARCHAR(50) NOT NULL, -- 'emergency_access', 'wallet_recovery', 'account_reset'
    
    is_used BOOLEAN DEFAULT FALSE,
    used_at TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    
    usage_restrictions JSONB -- What this code can be used for
);

-- =============================================
-- PARTITIONING FOR LARGE TABLES
-- =============================================

-- Partition operations table by month for better performance
CREATE TABLE wallet_operations_template (
    LIKE wallet_operations INCLUDING ALL
) PARTITION BY RANGE (created_at);

-- Create monthly partitions (example for 2024)
CREATE TABLE wallet_operations_2024_01 PARTITION OF wallet_operations_template
FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

CREATE TABLE wallet_operations_2024_02 PARTITION OF wallet_operations_template
FOR VALUES FROM ('2024-02-01') TO ('2024-03-01');

-- Similar partitioning for security events
CREATE TABLE security_events_template (
    LIKE security_events INCLUDING ALL
) PARTITION BY RANGE (created_at);

-- =============================================
-- VIEWS FOR COMMON QUERIES
-- =============================================

-- Active delegation sessions with user info
CREATE VIEW active_delegations AS
SELECT 
    ds.session_id,
    ds.user_wallet_address,
    ds.espace_address,
    ds.core_address,
    ds.expires_at,
    ds.last_activity,
    ds.config,
    u.email as user_email,
    u.username,
    EXTRACT(EPOCH FROM (ds.expires_at - NOW())) / 60 as minutes_until_expiry
FROM delegation_sessions ds
JOIN users u ON ds.user_id = u.user_id
WHERE ds.status = 'active' 
AND ds.expires_at > NOW();

-- User operation summary
CREATE VIEW user_operation_summary AS
SELECT 
    wo.user_id,
    u.email,
    COUNT(*) as total_operations,
    COUNT(*) FILTER (WHERE wo.status = 'completed') as completed_operations,
    COUNT(*) FILTER (WHERE wo.status = 'failed') as failed_operations,
    COUNT(*) FILTER (WHERE wo.chain_type = 'eSpace') as espace_operations,
    COUNT(*) FILTER (WHERE wo.chain_type = 'core') as core_operations,
    AVG(wo.gas_used) as avg_gas_used,
    MAX(wo.created_at) as last_operation_at
FROM wallet_operations wo
JOIN users u ON wo.user_id = u.user_id
GROUP BY wo.user_id, u.email;

-- Security alerts summary
CREATE VIEW security_alerts_summary AS
SELECT 
    se.user_id,
    u.email,
    COUNT(*) FILTER (WHERE se.severity = 'critical') as critical_alerts,
    COUNT(*) FILTER (WHERE se.severity = 'high') as high_alerts,
    COUNT(*) FILTER (WHERE se.severity = 'medium') as medium_alerts,
    COUNT(*) FILTER (WHERE se.resolution_status = 'open') as open_alerts,
    MAX(se.created_at) as last_alert_at
FROM security_events se
JOIN users u ON se.user_id = u.user_id
WHERE se.created_at > NOW() - INTERVAL '30 days'
GROUP BY se.user_id, u.email;
