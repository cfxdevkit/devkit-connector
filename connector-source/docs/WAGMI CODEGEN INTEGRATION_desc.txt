// connectors/conflux-delegation-connector.ts
import { Connector } from 'wagmi'
import { getAddress, type Address, type Chain } from 'viem'
import { confluxESpace, confluxCore } from '../chains/conflux'
import { ConfluxDelegationManager } from '../services/delegation-manager'

export interface ConfluxDelegationOptions {
  mode: 'server-managed' | 'user-delegated'
  serverUrl?: string
  delegationConfig?: DelegationConfig
  chains?: Chain[]
}

export class ConfluxDelegationConnector extends Connector<
  ConfluxDelegationProvider,
  ConfluxDelegationOptions
> {
  readonly id = 'conflux-delegation'
  readonly name = 'Conflux Delegation'
  readonly ready = true

  private delegationManager: ConfluxDelegationManager
  private currentMode: 'server-managed' | 'user-delegated' | null = null

  constructor(config: { chains?: Chain[]; options: ConfluxDelegationOptions }) {
    super({
      chains: config.chains ?? [confluxESpace, confluxCore],
      options: config.options,
    })
    
    this.delegationManager = new ConfluxDelegationManager({
      serverUrl: config.options.serverUrl,
      mode: config.options.mode,
    })
  }

  async connect(config?: { chainId?: number }): Promise<{
    account: Address
    chain: { id: number; unsupported: boolean }
    provider: ConfluxDelegationProvider
  }> {
    const { mode } = this.options
    
    if (mode === 'server-managed') {
      return this.connectServerManaged(config)
    } else {
      return this.connectUserDelegated(config)
    }
  }

  private async connectServerManaged(config?: { chainId?: number }) {
    const walletData = await this.delegationManager.loadServerWallet()
    
    const provider = new ConfluxDelegationProvider({
      mode: 'server-managed',
      addresses: walletData.addresses,
      walletId: walletData.walletId,
    })

    this.currentMode = 'server-managed'
    
    return {
      account: getAddress(walletData.addresses.eSpace),
      chain: { 
        id: config?.chainId ?? confluxESpace.id, 
        unsupported: false 
      },
      provider,
    }
  }

  private async connectUserDelegated(config?: { chainId?: number }) {
    // First connect user's wallet
    const userWallet = await this.requestUserWallet()
    
    // Create delegation session
    const sessionData = await this.delegationManager.createDelegationSession({
      userAddress: userWallet.address,
      config: this.options.delegationConfig,
    })

    const provider = new ConfluxDelegationProvider({
      mode: 'user-delegated',
      addresses: sessionData.addresses,
      sessionId: sessionData.sessionId,
      userWallet,
    })

    this.currentMode = 'user-delegated'

    return {
      account: getAddress(sessionData.addresses.eSpace),
      chain: { 
        id: config?.chainId ?? confluxESpace.id, 
        unsupported: false 
      },
      provider,
    }
  }

  async disconnect(): Promise<void> {
    if (this.currentMode === 'user-delegated') {
      await this.delegationManager.revokeDelegation()
    }
    this.currentMode = null
  }

  async getAccount(): Promise<Address> {
    const provider = await this.getProvider()
    return provider.getAccount()
  }

  async getChainId(): Promise<number> {
    const provider = await this.getProvider()
    return provider.getChainId()
  }

  async getProvider(): Promise<ConfluxDelegationProvider> {
    if (!this.currentMode) {
      throw new Error('Not connected')
    }
    return this.delegationManager.getProvider()
  }

  async isAuthorized(): Promise<boolean> {
    return this.currentMode !== null
  }

  protected onAccountsChanged(accounts: string[]): void {
    if (accounts.length === 0) this.emit('disconnect')
    else this.emit('change', { account: getAddress(accounts[0]!) })
  }

  protected onChainChanged(chain: string | number): void {
    const id = Number(chain)
    const unsupported = !this.chains.some((x) => x.id === id)
    this.emit('change', { chain: { id, unsupported } })
  }

  protected onDisconnect(): void {
    this.emit('disconnect')
  }
}

// Generated hook for writing with delegation support
export function useConfluxDelegationManagerWrite<
  TFunctionName extends string,
  TMode extends WriteContractMode = undefined,
  TChainId extends number = keyof typeof confluxDelegationManagerAddress,
>(
  config: UseContractWriteConfig<
    typeof confluxDelegationManagerABI,
    TFunctionName,
    TMode,
    TChainId
  > & {
    delegationMode?: 'server-managed' | 'user-delegated'
    requiresApproval?: boolean
    autoApprovalRules?: AutoApprovalRule[]
  } = {} as any,
) {
  const { mode, sendTransaction, addresses } = useConfluxDelegation()
  
  const baseHook = useContractWrite({
    address: confluxDelegationManagerAddress[config.chainId ?? confluxESpace.id],
    abi: confluxDelegationManagerABI,
    ...config,
  })

  // Override write function to use delegation
  const enhancedWrite = async (args: any) => {
    if (mode === 'user-delegated' || mode === 'server-managed') {
      // Use delegation system for transaction
      return sendTransaction({
        to: confluxDelegationManagerAddress[config.chainId ?? confluxESpace.id],
        data: encodeFunctionData({
          abi: confluxDelegationManagerABI,
          functionName: config.functionName!,
          args,
        }),
        requiresApproval: config.requiresApproval,
        autoApprovalRules: config.autoApprovalRules,
      })
    } else {
      // Fallback to normal wagmi behavior
      return baseHook.write?.(args)
    }
  }

  return {
    ...baseHook,
    write: enhancedWrite,
    writeAsync: enhancedWrite,
  }
}

  // Auto-generated, type-safe hooks with delegation
  const { data: allowance } = useConfluxSwapRouterWithDelegation({
    functionName: 'allowance',

    functionName: 'approve',

        maxValue: '1000000000000000000', // 1 token
        timeRestriction: {
          type: 'daily_window',
          allowedTimes: { startTime: '09:00', endTime: '17:00' }
        }
      })
    ]
  })

  const { write: swapTokens } = useConfluxSwapRouterWithDelegation({
    functionName: 'swapExactTokensForTokens',
    delegationMode: 'user-delegated',
    requiresApproval: true, // Always require approval for swaps
    autoApprovalRules: [
      // Auto-approve small swaps during business hours
      {
        type: 'transaction_value',
        condition: 'less_than',
        value: '100000000000000000000', // 100 tokens
        autoApprove: true,
        timeRestriction: {
          type: 'daily_window',
          allowedTimes: { startTime: '09:00', endTime: '17:00' }
        }
      }
    ]
  })

  return (
    <div>
      <h3>DeFi Dapp with Auto-Generated Delegation</h3>
      
      <div>
        <p>Current Allowance: {allowance?.toString()}</p>
        <button onClick={() => approveTokens?.()}>
          Approve Tokens (Auto-approved if under 1 token)
        </button>
      </div>

      <div>
        <button onClick={() => swapTokens?.(['1000000000000000000', '0', ['0x...', '0x...'], '0x...', Date.now() + 300000])}>
          Swap Tokens (Auto-approved if under 100 tokens during business hours)
        </button>
      </div>
    </div>
  )
}
