// types/delegation-config.ts
export interface WalletDelegationConfig {
  // === SESSION MANAGEMENT ===
  sessionDuration?: number;              // Duration in minutes (default: 60)
  maxIdleTime?: number;                  // Auto-expire after inactivity (minutes)
  autoRenew?: boolean;                   // Auto-renew before expiration
  maxRenewals?: number;                  // Maximum number of renewals
  
  // === OPERATION PERMISSIONS ===
  allowedOperations?: OperationType[];   // Which operations are permitted
  blockedOperations?: OperationType[];   // Explicitly blocked operations
  requireApprovalFor?: OperationType[];  // Operations requiring manual approval
  
  // === TRANSACTION LIMITS ===
  transactionLimits?: {
    maxValue?: string;                   // Max transaction value (in wei)
    maxGasPrice?: string;                // Max gas price willing to pay
    maxGasLimit?: string;                // Max gas limit per transaction
    maxDailyValue?: string;              // Max total value per day
    maxDailyCount?: number;              // Max transactions per day
    maxHourlyCount?: number;             // Max transactions per hour
  };
  
  // === CONTRACT INTERACTION ===
  contractPermissions?: {
    allowedContracts?: string[];         // Whitelist of contract addresses
    blockedContracts?: string[];         // Blacklist of contract addresses
    allowedFunctions?: FunctionSelector[]; // Specific function signatures
    blockedFunctions?: FunctionSelector[]; // Blocked function signatures
    requireApprovalForContracts?: string[]; // Contracts requiring approval
  };
  
  // === CHAIN SPECIFIC SETTINGS ===
  chainSettings?: {
    allowedChains?: ChainType[];         // Which chains can be used
    preferredChain?: ChainType;          // Default chain for operations
    crossChainOperations?: boolean;      // Allow cross-chain transactions
    autoSwitchChain?: boolean;           // Auto-switch chains as needed
  };
  
  // === AUTO-APPROVAL RULES ===
  autoApprovalRules?: AutoApprovalRule[];
  
  // === SECURITY SETTINGS ===
  securitySettings?: {
    requireSignatureFor?: OperationType[]; // Operations requiring fresh signature
    ipWhitelist?: string[];              // Allowed IP addresses
    userAgentValidation?: boolean;       // Validate user agent
    timeBasedRestrictions?: TimeRestriction[]; // Time-based limitations
    multiSigRequired?: boolean;          // Require additional signatures
  };
  
  // === NOTIFICATION PREFERENCES ===
  notifications?: {
    realTimeUpdates?: boolean;           // Real-time WebSocket updates
    emailNotifications?: boolean;        // Email for important operations
    webhookUrl?: string;                 // Webhook for external notifications
    notifyOnOperations?: OperationType[]; // Which operations to notify about
    alertThresholds?: AlertThreshold[];  // Custom alert thresholds
  };
  
  // === RECOVERY & EMERGENCY ===
  emergencySettings?: {
    emergencyContacts?: string[];        // Emergency wallet addresses
    killSwitchEnabled?: boolean;         // Enable emergency stop
    recoverySeed?: string;               // Recovery mechanism identifier
    timeoutRecovery?: number;            // Auto-recovery after timeout
  };
  
  // === METADATA & CONTEXT ===
  metadata?: {
    delegationPurpose?: string;          // Why this delegation is created
    expectedUsage?: string;              // Expected usage pattern
    tags?: string[];                     // Organizational tags
    externalRef?: string;                // External reference ID
  };
}

// Supporting types
export type OperationType = 
  | 'sign_transaction' 
  | 'sign_message' 
  | 'get_balance' 
  | 'estimate_gas'
  | 'deploy_contract'
  | 'interact_contract'
  | 'transfer_tokens'
  | 'swap_tokens'
  | 'stake_tokens'
  | 'unstake_tokens'
  | 'bridge_tokens';

export type ChainType = 'eSpace' | 'core';

export interface FunctionSelector {
  signature: string;                     // Function signature (e.g., "transfer(address,uint256)")
  contractAddress?: string;              // Specific contract (optional)
}

export interface AutoApprovalRule {
  id?: string;                          // Rule identifier
  name?: string;                        // Human-readable name
  description?: string;                 // Rule description
  type: 'transaction_value' | 'contract_address' | 'function_signature' | 'gas_price' | 'recipient' | 'time_based';
  condition: 'less_than' | 'greater_than' | 'equals' | 'in_range' | 'whitelist' | 'blacklist' | 'matches_pattern';
  value: string | string[] | number | { min: string; max: string };
  autoApprove: boolean;                 // Whether to auto-approve or auto-reject
  priority?: number;                    // Rule priority (higher = evaluated first)
  chainSpecific?: ChainType;            // Apply only to specific chain
}

export interface TimeRestriction {
  type: 'daily_window' | 'weekly_schedule' | 'specific_dates';
  allowedTimes?: {
    startTime: string;                  // "09:00"
    endTime: string;                    // "17:00"
    timezone?: string;                  // "UTC", "America/New_York"
    daysOfWeek?: number[];              // [1,2,3,4,5] for Monday-Friday
  };
  blockedDates?: string[];              // ["2024-12-25", "2024-01-01"]
}

export interface AlertThreshold {
  type: 'transaction_value' | 'daily_volume' | 'operation_count' | 'gas_consumption';
  threshold: string | number;
  action: 'notify' | 'require_approval' | 'block';
  message?: string;
}

// === CONFIGURATION BUILDER ===
export class DelegationConfigBuilder {
  private config: WalletDelegationConfig = {};

  // Session management
  withSession(duration: number, autoRenew = false): this {
    this.config.sessionDuration = duration;
    this.config.autoRenew = autoRenew;
    return this;
  }

  // Operation permissions
  allowOperations(...operations: OperationType[]): this {
    this.config.allowedOperations = operations;
    return this;
  }

  requireApprovalFor(...operations: OperationType[]): this {
    this.config.requireApprovalFor = operations;
    return this;
  }

  // Transaction limits
  withTransactionLimits(limits: Partial<WalletDelegationConfig['transactionLimits']>): this {
    this.config.transactionLimits = { ...this.config.transactionLimits, ...limits };
    return this;
  }

  // Contract permissions
  allowContracts(...addresses: string[]): this {
    if (!this.config.contractPermissions) this.config.contractPermissions = {};
    this.config.contractPermissions.allowedContracts = addresses;
    return this;
  }

  blockContracts(...addresses: string[]): this {
    if (!this.config.contractPermissions) this.config.contractPermissions = {};
    this.config.contractPermissions.blockedContracts = addresses;
    return this;
  }

  // Auto-approval rules
  addAutoApprovalRule(rule: AutoApprovalRule): this {
    if (!this.config.autoApprovalRules) this.config.autoApprovalRules = [];
    this.config.autoApprovalRules.push(rule);
    return this;
  }

  // Security settings
  withSecurity(settings: Partial<WalletDelegationConfig['securitySettings']>): this {
    this.config.securitySettings = { ...this.config.securitySettings, ...settings };
    return this;
  }

  // Notifications
  withNotifications(settings: Partial<WalletDelegationConfig['notifications']>): this {
    this.config.notifications = { ...this.config.notifications, ...settings };
    return this;
  }

  // Build final config
  build(): WalletDelegationConfig {
    return { ...this.config };
  }

  // Load from preset
  static fromPreset(preset: keyof typeof DelegationPresets): DelegationConfigBuilder {
    const builder = new DelegationConfigBuilder();
    builder.config = { ...DelegationPresets[preset] };
    return builder;
  }
}

// === VALIDATION FUNCTIONS ===
export function validateDelegationConfig(config: WalletDelegationConfig): {
  isValid: boolean;
  errors: string[];
  warnings: string[];
} {
  const errors: string[] = [];
  const warnings: string[] = [];

  // Validate session duration
  if (config.sessionDuration && config.sessionDuration > 1440) {
    warnings.push('Session duration over 24 hours may pose security risks');
  }

  // Validate transaction limits
  if (config.transactionLimits?.maxValue) {
    const maxValue = BigInt(config.transactionLimits.maxValue);
    if (maxValue > BigInt('1000000000000000000000')) { // 1000 CFX
      warnings.push('High transaction limit detected');
    }
  }

  // Validate contract addresses
  if (config.contractPermissions?.allowedContracts) {
    for (const address of config.contractPermissions.allowedContracts) {
      if (!address.startsWith('0x') || address.length !== 42) {
        errors.push(`Invalid contract address: ${address}`);
      }
    }
  }

  // Validate auto-approval rules
  if (config.autoApprovalRules) {
    for (const rule of config.autoApprovalRules) {
      if (rule.type === 'transaction_value' && typeof rule.value !== 'string') {
        errors.push('Transaction value rules must have string values');
      }
    }
  }

  return {
    isValid: errors.length === 0,
    errors,
    warnings
  };
}

// === REACT HOOK FOR CONFIGURATION ===
export function useDelegationConfig() {
  const [config, setConfig] = useState<WalletDelegationConfig>({});
  const [preset, setPreset] = useState<keyof typeof DelegationPresets | null>(null);

  const loadPreset = (presetName: keyof typeof DelegationPresets) => {
    setConfig(DelegationPresets[presetName]);
    setPreset(presetName);
  };

  const updateConfig = (updates: Partial<WalletDelegationConfig>) => {
    setConfig(prev => ({ ...prev, ...updates }));
    setPreset(null); // Clear preset when manually editing
  };

  const resetConfig = () => {
    setConfig({});
    setPreset(null);
  };

  const validation = validateDelegationConfig(config);

  return {
    config,
    preset,
    loadPreset,
    updateConfig,
    resetConfig,
    validation,
    builder: () => new DelegationConfigBuilder()
  };
}
