// types/delegation-types.ts
export interface DelegationOperation {
  id: string;
  sessionId: string;
  type: 'sign_transaction' | 'sign_message' | 'get_balance' | 'estimate_gas';
  status: 'pending' | 'approved' | 'rejected' | 'completed' | 'failed';
  chainType: 'eSpace' | 'core';
  payload: any;
  result?: any;
  error?: string;
  createdAt: number;
  completedAt?: number;
  requiresUserApproval?: boolean;
}

export interface DelegationEvent {
  id: string;
  sessionId: string;
  type: 'session_created' | 'operation_requested' | 'operation_completed' | 'session_expired' | 'permission_denied';
  data: any;
  timestamp: number;
}

export interface OperationLimits {
  maxTransactionValue: string;
  maxDailyTransactions: number;
  allowedContracts?: string[];
  blockedContracts?: string[];
}

export interface EnhancedDelegationSession extends DelegatedSession {
  limits: OperationLimits;
  operationCount: number;
  dailyVolume: string;
  lastActivity: number;
  autoApprovalRules: AutoApprovalRule[];
}

export interface AutoApprovalRule {
  type: 'transaction_value' | 'contract_address' | 'function_signature';
  condition: 'less_than' | 'equals' | 'whitelist';
  value: string | string[];
  autoApprove: boolean;
}

// server/delegation-event-manager.ts
import { EventEmitter } from 'events';

export class DelegationEventManager extends EventEmitter {
  private static instance: DelegationEventManager;
  private events: Map<string, DelegationEvent[]> = new Map();
  
  static getInstance(): DelegationEventManager {
    if (!this.instance) {
      this.instance = new DelegationEventManager();
    }
    return this.instance;
  }

  /**
   * Emit and store delegation event
   */
  emitDelegationEvent(sessionId: string, type: string, data: any): void {
    const event: DelegationEvent = {
      id: `evt_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
      sessionId,
      type: type as any,
      data,
      timestamp: Date.now(),
    };

    // Store event
    if (!this.events.has(sessionId)) {
      this.events.set(sessionId, []);
    }
    this.events.get(sessionId)!.push(event);

    // Emit for real-time listeners
    this.emit('delegation_event', event);
    this.emit(`session_${sessionId}`, event);
    
    // Cleanup old events (keep last 100 per session)
    const sessionEvents = this.events.get(sessionId)!;
    if (sessionEvents.length > 100) {
      this.events.set(sessionId, sessionEvents.slice(-100));
    }
  }

  /**
   * Get events for a session
   */
  getSessionEvents(sessionId: string): DelegationEvent[] {
    return this.events.get(sessionId) || [];
  }

  /**
   * Clear events for a session
   */
  clearSessionEvents(sessionId: string): void {
    this.events.delete(sessionId);
  }
}

// server/enhanced-delegation-service.ts
import { WalletDelegationService } from './delegation-service';
import { DelegationEventManager } from './delegation-event-manager';

export class EnhancedDelegationService extends WalletDelegationService {
  private enhancedSessions: Map<string, EnhancedDelegationSession> = new Map();
  private operations: Map<string, DelegationOperation> = new Map();
  private eventManager: DelegationEventManager;
  
  constructor() {
    super();
    this.eventManager = DelegationEventManager.getInstance();
  }

  /**
   * Create enhanced delegation session with limits and rules
   */
  async createEnhancedDelegationSession(
    userAddress: string,
    signature: string,
    message: string,
    config: WalletDelegationConfig & {
      limits?: OperationLimits;
      autoApprovalRules?: AutoApprovalRule[];
    } = {}
  ): Promise<EnhancedDelegationSession> {
    const baseSession = await this.createDelegationSession(userAddress, signature, message, config);
    
    const enhancedSession: EnhancedDelegationSession = {
      ...baseSession,
      limits: config.limits || {
        maxTransactionValue: '10000000000000000000', // 10 CFX
        maxDailyTransactions: 50,
      },
      operationCount: 0,
      dailyVolume: '0',
      lastActivity: Date.now(),
      autoApprovalRules: config.autoApprovalRules || [],
    };

    this.enhancedSessions.set(baseSession.sessionId, enhancedSession);
    
    this.eventManager.emitDelegationEvent(baseSession.sessionId, 'session_created', {
      session: enhancedSession,
      limits: enhancedSession.limits,
    });

    return enhancedSession;
  }

  /**
   * Request operation with approval workflow
   */
  async requestOperation(
    sessionId: string,
    operationType: DelegationOperation['type'],
    payload: any,
    chainType: 'eSpace' | 'core'
  ): Promise<DelegationOperation> {
    const session = this.getEnhancedSession(sessionId);
    if (!session) {
      throw new Error('Invalid or expired session');
    }

    // Create operation
    const operation: DelegationOperation = {
      id: `op_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
      sessionId,
      type: operationType,
      status: 'pending',
      chainType,
      payload,
      createdAt: Date.now(),
      requiresUserApproval: await this.requiresUserApproval(session, operationType, payload),
    };

    this.operations.set(operation.id, operation);

    // Update session activity
    session.lastActivity = Date.now();
    session.operationCount++;

    this.eventManager.emitDelegationEvent(sessionId, 'operation_requested', {
      operation,
      requiresApproval: operation.requiresUserApproval,
    });

    // Auto-approve if rules allow
    if (!operation.requiresUserApproval) {
      operation.status = 'approved';
      this.eventManager.emitDelegationEvent(sessionId, 'operation_auto_approved', { operation });
    }

    return operation;
  }

  /**
   * Execute approved operation
   */
  async executeOperation(operationId: string): Promise<any> {
    const operation = this.operations.get(operationId);
    if (!operation || operation.status !== 'approved') {
      throw new Error('Operation not found or not approved');
    }

    const session = this.getEnhancedSession(operation.sessionId);
    if (!session) {
      throw new Error('Session expired');
    }

    try {
      operation.status = 'pending';
      
      let result: any;
      
      switch (operation.type) {
        case 'sign_transaction':
          result = await this.executeSignTransaction(session, operation);
          break;
        case 'sign_message':
          result = await this.executeSignMessage(session, operation);
          break;
        case 'get_balance':
          result = await this.executeGetBalance(session, operation);
          break;
        case 'estimate_gas':
          result = await this.executeEstimateGas(session, operation);
          break;
        default:
          throw new Error(`Unsupported operation type: ${operation.type}`);
      }

      operation.status = 'completed';
      operation.result = result;
      operation.completedAt = Date.now();

      // Update session limits
      await this.updateSessionLimits(session, operation);

      this.eventManager.emitDelegationEvent(operation.sessionId, 'operation_completed', {
        operation,
        result,
      });

      return result;
    } catch (error) {
      operation.status = 'failed';
      operation.error = error.message;
      operation.completedAt = Date.now();

      this.eventManager.emitDelegationEvent(operation.sessionId, 'operation_failed', {
        operation,
        error: error.message,
      });

      throw error;
    }
  }

  /**
   * Approve operation (user consent)
   */
  async approveOperation(operationId: string, userApproval: boolean): Promise<void> {
    const operation = this.operations.get(operationId);
    if (!operation) {
      throw new Error('Operation not found');
    }

    operation.status = userApproval ? 'approved' : 'rejected';
    
    this.eventManager.emitDelegationEvent(operation.sessionId, userApproval ? 'operation_approved' : 'operation_rejected', {
      operation,
    });

    if (userApproval) {
      // Auto-execute after approval
      await this.executeOperation(operationId);
    }
  }

  /**
   * Get enhanced session
   */
  getEnhancedSession(sessionId: string): EnhancedDelegationSession | null {
    const session = this.enhancedSessions.get(sessionId);
    if (!session || session.expiresAt < Date.now()) {
      this.enhancedSessions.delete(sessionId);
      this.eventManager.emitDelegationEvent(sessionId, 'session_expired', { sessionId });
      return null;
    }
    return session;
  }

  /**
   * Get operations for session
   */
  getSessionOperations(sessionId: string): DelegationOperation[] {
    return Array.from(this.operations.values())
      .filter(op => op.sessionId === sessionId)
      .sort((a, b) => b.createdAt - a.createdAt);
  }

  /**
   * Check if operation requires user approval
   */
  private async requiresUserApproval(
    session: EnhancedDelegationSession,
    operationType: string,
    payload: any
  ): Promise<boolean> {
    if (operationType === 'get_balance' || operationType === 'estimate_gas') {
      return false; // Read operations don't need approval
    }

    if (operationType === 'sign_transaction') {
      // Check auto-approval rules
      for (const rule of session.autoApprovalRules) {
        if (await this.matchesAutoApprovalRule(rule, payload)) {
          return !rule.autoApprove;
        }
      }

      // Check limits
      const transactionValue = BigInt(payload.value || '0');
      const maxValue = BigInt(session.limits.maxTransactionValue);
      
      if (transactionValue > maxValue) {
        return true; // Requires approval for large transactions
      }

      // Check daily limits
      if (session.operationCount >= session.limits.maxDailyTransactions) {
        return true;
      }
    }

    return false; // Default to auto-approve
  }

  /**
   * Check if payload matches auto-approval rule
   */
  private async matchesAutoApprovalRule(rule: AutoApprovalRule, payload: any): Promise<boolean> {
    switch (rule.type) {
      case 'transaction_value':
        const value = BigInt(payload.value || '0');
        const ruleValue = BigInt(rule.value as string);
        return rule.condition === 'less_than' ? value < ruleValue : value === ruleValue;
        
      case 'contract_address':
        const to = payload.to?.toLowerCase();
        const addresses = (rule.value as string[]).map(addr => addr.toLowerCase());
        return rule.condition === 'whitelist' ? addresses.includes(to) : !addresses.includes(to);
        
      default:
        return false;
    }
  }

  /**
   * Execute different operation types
   */
  private async executeSignTransaction(session: EnhancedDelegationSession, operation: DelegationOperation): Promise<string> {
    // This would integrate with your existing signing logic
    // For demonstration, returning a mock signed transaction
    return `0x${Math.random().toString(16).substr(2, 64)}`;
  }

  private async executeSignMessage(session: EnhancedDelegationSession, operation: DelegationOperation): Promise<string> {
    // Mock message signing
    return `0x${Math.random().toString(16).substr(2, 130)}`;
  }

  private async executeGetBalance(session: EnhancedDelegationSession, operation: DelegationOperation): Promise<string> {
    // Mock balance retrieval
    return '1000000000000000000'; // 1 CFX
  }

  private async executeEstimateGas(session: EnhancedDelegationSession, operation: DelegationOperation): Promise<string> {
    // Mock gas estimation
    return '21000';
  }

  /**
   * Update session limits after operation
   */
  private async updateSessionLimits(session: EnhancedDelegationSession, operation: DelegationOperation): Promise<void> {
    if (operation.type === 'sign_transaction' && operation.payload.value) {
      const currentVolume = BigInt(session.dailyVolume);
      const transactionValue = BigInt(operation.payload.value);
      session.dailyVolume = (currentVolume + transactionValue).toString();
    }
  }

  /**
   * Get session analytics
   */
  getSessionAnalytics(sessionId: string): {
    totalOperations: number;
    successfulOperations: number;
    failedOperations: number;
    totalVolume: string;
    averageResponseTime: number;
  } {
    const operations = this.getSessionOperations(sessionId);
    const session = this.getEnhancedSession(sessionId);

    const successful = operations.filter(op => op.status === 'completed').length;
    const failed = operations.filter(op => op.status === 'failed').length;
    
    const avgResponseTime = operations
      .filter(op => op.completedAt)
      .reduce((sum, op) => sum + (op.completedAt! - op.createdAt), 0) / operations.length || 0;

    return {
      totalOperations: operations.length,
      successfulOperations: successful,
      failedOperations: failed,
      totalVolume: session?.dailyVolume || '0',
      averageResponseTime: Math.round(avgResponseTime),
    };
  }
}

// server/api/delegation-events/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { EnhancedDelegationService } from '../../enhanced-delegation-service';
import { DelegationEventManager } from '../../delegation-event-manager';

const delegationService = new EnhancedDelegationService();
const eventManager = DelegationEventManager.getInstance();

export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url);
  const sessionId = searchParams.get('sessionId');
  const action = searchParams.get('action');

  if (!sessionId) {
    return NextResponse.json({ error: 'Session ID required' }, { status: 400 });
  }

  try {
    switch (action) {
      case 'events':
        const events = eventManager.getSessionEvents(sessionId);
        return NextResponse.json({ events });

      case 'operations':
        const operations = delegationService.getSessionOperations(sessionId);
        return NextResponse.json({ operations });

      case 'analytics':
        const analytics = delegationService.getSessionAnalytics(sessionId);
        return NextResponse.json({ analytics });

      case 'status':
        const session = delegationService.getEnhancedSession(sessionId);
        return NextResponse.json({ 
          session: session ? {
            sessionId: session.sessionId,
            expiresAt: session.expiresAt,
            operationCount: session.operationCount,
            dailyVolume: session.dailyVolume,
            lastActivity: session.lastActivity,
          } : null 
        });

      default:
        return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
    }
  } catch (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

export async function POST(request: NextRequest) {
  try {
    const { action, sessionId, operationId, ...params } = await request.json();

    switch (action) {
      case 'approve_operation':
        const { approved } = params;
        await delegationService.approveOperation(operationId, approved);
        return NextResponse.json({ success: true });

      case 'request_operation':
        const { type, payload, chainType } = params;
        const operation = await delegationService.requestOperation(sessionId, type, payload, chainType);
        return NextResponse.json({ operation });

      case 'execute_operation':
        const result = await delegationService.executeOperation(operationId);
        return NextResponse.json({ result });

      default:
        return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
    }
  } catch (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

// client/hooks/use-delegation-manager.ts
'use client';

import { useState, useEffect, useCallback } from 'react';
import { useUnifiedWallet } from './use-unified-wallet';

interface DelegationManagerState {
  operations: DelegationOperation[];
  events: DelegationEvent[];
  analytics: any;
  isLoading: boolean;
  pendingOperations: DelegationOperation[];
}

export function useDelegationManager() {
  const { sessionId, mode } = useUnifiedWallet();
  const [state, setState] = useState<DelegationManagerState>({
    operations: [],
    events: [],
    analytics: null,
    isLoading: false,
    pendingOperations: [],
  });

  const fetchDelegationData = useCallback(async () => {
    if (!sessionId || mode !== 'user-delegated') return;

    setState(prev => ({ ...prev, isLoading: true }));

    try {
      const [operationsRes, eventsRes, analyticsRes] = await Promise.all([
        fetch(`/api/delegation-events?sessionId=${sessionId}&action=operations`),
        fetch(`/api/delegation-events?sessionId=${sessionId}&action=events`),
        fetch(`/api/delegation-events?sessionId=${sessionId}&action=analytics`),
      ]);

      const operations = await operationsRes.json();
      const events = await eventsRes.json();
      const analytics = await analyticsRes.json();

      setState({
        operations: operations.operations || [],
        events: events.events || [],
        analytics: analytics.analytics,
        isLoading: false,
        pendingOperations: (operations.operations || []).filter((op: DelegationOperation) => 
          op.status === 'pending' && op.requiresUserApproval
        ),
      });
    } catch (error) {
      console.error('Failed to fetch delegation data:', error);
      setState(prev => ({ ...prev, isLoading: false }));
    }
  }, [sessionId, mode]);

  const approveOperation = useCallback(async (operationId: string, approved: boolean) => {
    try {
      await fetch('/api/delegation-events', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          action: 'approve_operation',
          operationId,
          approved,
        }),
      });

      // Refresh data after approval
      await fetchDelegationData();
    } catch (error) {
      console.error('Failed to approve operation:', error);
    }
  }, [fetchDelegationData]);

  // Auto-refresh every 5 seconds
  useEffect(() => {
    if (sessionId && mode === 'user-delegated') {
      fetchDelegationData();
      const interval = setInterval(fetchDelegationData, 5000);
      return () => clearInterval(interval);
    }
  }, [sessionId, mode, fetchDelegationData]);

  return {
    ...state,
    approveOperation,
    refresh: fetchDelegationData,
  };
}
