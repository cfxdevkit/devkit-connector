// types/wallet-options.ts
export type WalletMode = 'server-managed' | 'user-delegated';

export interface WalletDelegationConfig {
  sessionDuration?: number; // in minutes
  allowedOperations?: string[];
  maxTransactionValue?: string;
}

export interface DelegatedSession {
  sessionId: string;
  userAddress: string;
  eSpaceAddress: string;
  coreAddress: string;
  expiresAt: number;
  permissions: string[];
}

// server/delegation-service.ts
import { randomBytes, createHash } from 'crypto';
import jwt from 'jsonwebtoken';

export class WalletDelegationService {
  private static instance: WalletDelegationService;
  private activeSessions: Map<string, DelegatedSession> = new Map();
  
  static getInstance(): WalletDelegationService {
    if (!this.instance) {
      this.instance = new WalletDelegationService();
    }
    return this.instance;
  }

  /**
   * Create a delegation session for user's wallet
   */
  async createDelegationSession(
    userAddress: string,
    signature: string,
    message: string,
    config: WalletDelegationConfig = {}
  ): Promise<DelegatedSession> {
    // Verify the signature to prove wallet ownership
    const isValidSignature = await this.verifySignature(userAddress, message, signature);
    if (!isValidSignature) {
      throw new Error('Invalid signature - wallet ownership not proven');
    }

    const sessionId = randomBytes(32).toString('hex');
    const expiresAt = Date.now() + (config.sessionDuration || 60) * 60 * 1000; // Default 1 hour
    
    // Convert user's eSpace address to Core format
    const eSpaceAddress = userAddress;
    const coreAddress = format.address(userAddress, 1029);

    const session: DelegatedSession = {
      sessionId,
      userAddress,
      eSpaceAddress,
      coreAddress,
      expiresAt,
      permissions: config.allowedOperations || ['sign_transaction', 'sign_message'],
    };

    this.activeSessions.set(sessionId, session);

    // Set cleanup timer
    setTimeout(() => {
      this.activeSessions.delete(sessionId);
    }, expiresAt - Date.now());

    return session;
  }

  /**
   * Get active delegation session
   */
  getDelegationSession(sessionId: string): DelegatedSession | null {
    const session = this.activeSessions.get(sessionId);
    if (!session || session.expiresAt < Date.now()) {
      this.activeSessions.delete(sessionId);
      return null;
    }
    return session;
  }

  /**
   * Verify wallet signature for delegation
   */
  private async verifySignature(address: string, message: string, signature: string): Promise<boolean> {
    try {
      // Use ethers or viem to verify signature
      const { verifyMessage } = await import('ethers');
      const recoveredAddress = verifyMessage(message, signature);
      return recoveredAddress.toLowerCase() === address.toLowerCase();
    } catch {
      return false;
    }
  }

  /**
   * Revoke delegation session
   */
  revokeDelegation(sessionId: string): boolean {
    return this.activeSessions.delete(sessionId);
  }

  /**
   * Check if operation is allowed for session
   */
  isOperationAllowed(sessionId: string, operation: string): boolean {
    const session = this.getDelegationSession(sessionId);
    return session ? session.permissions.includes(operation) : false;
  }
}

// server/unified-wallet-service.ts
import { ServerWalletService } from './wallet-service'; // From Option A
import { WalletDelegationService } from './delegation-service';

export class UnifiedWalletService {
  private serverWalletService: ServerWalletService;
  private delegationService: WalletDelegationService;

  constructor() {
    this.serverWalletService = ServerWalletService.getInstance();
    this.delegationService = WalletDelegationService.getInstance();
  }

  /**
   * Load wallet based on mode (Option A or B)
   */
  async loadWallet(userId: string, mode: WalletMode, sessionId?: string) {
    if (mode === 'server-managed') {
      // Option A: Server manages the mnemonic
      return await this.serverWalletService.loadWalletFromServer(userId);
    } else {
      // Option B: User delegated wallet
      if (!sessionId) {
        throw new Error('Session ID required for delegated wallet');
      }
      
      const session = this.delegationService.getDelegationSession(sessionId);
      if (!session) {
        throw new Error('Invalid or expired delegation session');
      }

      return {
        eSpaceAddress: session.eSpaceAddress,
        coreAddress: session.coreAddress,
        walletId: sessionId,
        mode: 'delegated' as const,
      };
    }
  }

  /**
   * Sign transaction with appropriate method
   */
  async signTransaction(
    walletId: string, 
    transaction: any, 
    chainType: 'eSpace' | 'core',
    mode: WalletMode
  ): Promise<string> {
    if (mode === 'server-managed') {
      // Option A: Use server's private key
      return await this.serverWalletService.signTransaction(walletId, transaction, chainType);
    } else {
      // Option B: Delegate back to user's wallet for signing
      // Server validates and forwards to client for signing
      const session = this.delegationService.getDelegationSession(walletId);
      if (!session) {
        throw new Error('Invalid delegation session');
      }

      if (!this.delegationService.isOperationAllowed(walletId, 'sign_transaction')) {
        throw new Error('Transaction signing not permitted in this session');
      }

      // Return a pending transaction that needs client-side signing
      return JSON.stringify({
        type: 'PENDING_SIGNATURE',
        sessionId: walletId,
        transaction,
        chainType,
        requiresClientSignature: true,
      });
    }
  }
}

// server/api/wallet-unified/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { UnifiedWalletService } from '../../unified-wallet-service';
import { WalletDelegationService } from '../../delegation-service';

const unifiedWalletService = new UnifiedWalletService();
const delegationService = WalletDelegationService.getInstance();

export async function POST(request: NextRequest) {
  try {
    const { action, mode, ...params } = await request.json();

    switch (action) {
      case 'create_delegation':
        const { userAddress, signature, message, config } = params;
        const session = await delegationService.createDelegationSession(
          userAddress, 
          signature, 
          message, 
          config
        );
        return NextResponse.json({ 
          sessionId: session.sessionId,
          expiresAt: session.expiresAt,
          addresses: {
            eSpace: session.eSpaceAddress,
            core: session.coreAddress,
          }
        });

      case 'load_wallet':
        const { userId, sessionId } = params;
        const walletData = await unifiedWalletService.loadWallet(userId, mode, sessionId);
        return NextResponse.json(walletData);

      case 'sign_transaction':
        const { walletId, transaction, chainType } = params;
        const signature = await unifiedWalletService.signTransaction(
          walletId, 
          transaction, 
          chainType, 
          mode
        );
        return NextResponse.json({ signature });

      case 'revoke_delegation':
        const { sessionId: revokeSessionId } = params;
        const revoked = delegationService.revokeDelegation(revokeSessionId);
        return NextResponse.json({ success: revoked });

      default:
        return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
    }
  } catch (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

// client/hooks/use-unified-wallet.ts
'use client';

import { useState, useCallback } from 'react';
import { useAccount, useSignMessage } from 'wagmi';
import { useConflux } from '../providers/conflux-server-provider';

interface UnifiedWalletState {
  mode: WalletMode | null;
  eSpaceAddress: string | null;
  coreAddress: string | null;
  walletId: string | null;
  sessionId: string | null;
  isLoading: boolean;
  error: string | null;
}

export function useUnifiedWallet() {
  const { address, isConnected } = useAccount(); // Wagmi for user's wallet
  const { signMessageAsync } = useSignMessage();
  const { currentChain, eSpaceAdapter, coreAdapter } = useConflux();
  
  const [walletState, setWalletState] = useState<UnifiedWalletState>({
    mode: null,
    eSpaceAddress: null,
    coreAddress: null,
    walletId: null,
    sessionId: null,
    isLoading: false,
    error: null,
  });

  const callWalletAPI = async (action: string, params: any = {}) => {
    const response = await fetch('/api/wallet-unified', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action, mode: walletState.mode, ...params }),
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error || 'API call failed');
    }

    return response.json();
  };

  /**
   * Option A: Load server-managed wallet
   */
  const loadServerWallet = useCallback(async (userId: string) => {
    setWalletState(prev => ({ ...prev, isLoading: true, error: null, mode: 'server-managed' }));
    
    try {
      const walletData = await callWalletAPI('load_wallet', { userId });
      setWalletState({
        mode: 'server-managed',
        eSpaceAddress: walletData.eSpaceAddress,
        coreAddress: walletData.coreAddress,
        walletId: walletData.walletId,
        sessionId: null,
        isLoading: false,
        error: null,
      });
    } catch (error) {
      setWalletState(prev => ({ ...prev, isLoading: false, error: error.message }));
    }
  }, []);

  /**
   * Option B: Delegate user's wallet to server
   */
  const delegateUserWallet = useCallback(async (config?: WalletDelegationConfig) => {
    if (!isConnected || !address) {
      throw new Error('Please connect your wallet first');
    }

    setWalletState(prev => ({ ...prev, isLoading: true, error: null, mode: 'user-delegated' }));

    try {
      // Create delegation message
      const message = `Delegate wallet operations to server\nAddress: ${address}\nTimestamp: ${Date.now()}`;
      
      // Sign message with user's wallet
      const signature = await signMessageAsync({ message });
      
      // Create delegation session on server
      const delegationData = await callWalletAPI('create_delegation', {
        userAddress: address,
        signature,
        message,
        config,
      });

      setWalletState({
        mode: 'user-delegated',
        eSpaceAddress: delegationData.addresses.eSpace,
        coreAddress: delegationData.addresses.core,
        walletId: delegationData.sessionId,
        sessionId: delegationData.sessionId,
        isLoading: false,
        error: null,
      });
    } catch (error) {
      setWalletState(prev => ({ ...prev, isLoading: false, error: error.message }));
    }
  }, [isConnected, address, signMessageAsync]);

  /**
   * Send transaction (works for both modes)
   */
  const sendTransaction = useCallback(async (transaction: any) => {
    if (!walletState.walletId) {
      throw new Error('No wallet loaded');
    }

    try {
      const { signature } = await callWalletAPI('sign_transaction', {
        walletId: walletState.walletId,
        transaction,
        chainType: currentChain,
      });

      // Handle different signature types
      if (typeof signature === 'string' && signature.startsWith('{')) {
        // Option B: Requires client-side signing
        const pendingTx = JSON.parse(signature);
        if (pendingTx.requiresClientSignature) {
          // Sign with user's wallet and broadcast
          const clientSignature = await signMessageAsync({ 
            message: JSON.stringify(pendingTx.transaction) 
          });
          
          // Broadcast using appropriate adapter
          if (currentChain === 'eSpace' && eSpaceAdapter) {
            return await eSpaceAdapter.sendTransaction({
              ...pendingTx.transaction,
              signature: clientSignature,
            });
          } else if (currentChain === 'core' && coreAdapter) {
            return await coreAdapter.sendTransaction({
              ...pendingTx.transaction,
              signature: clientSignature,
            });
          }
        }
      } else {
        // Option A: Server-signed transaction
        if (currentChain === 'eSpace' && eSpaceAdapter) {
          return await eSpaceAdapter.sendTransaction(transaction);
        } else if (currentChain === 'core' && coreAdapter) {
          return await coreAdapter.sendTransaction({ ...transaction, signature });
        }
      }

      throw new Error('No adapter available');
    } catch (error) {
      throw new Error(`Transaction failed: ${error.message}`);
    }
  }, [walletState.walletId, currentChain, eSpaceAdapter, coreAdapter, signMessageAsync, callWalletAPI]);

  /**
   * Clear wallet (works for both modes)
   */
  const clearWallet = useCallback(async () => {
    if (walletState.mode === 'user-delegated' && walletState.sessionId) {
      await callWalletAPI('revoke_delegation', { sessionId: walletState.sessionId });
    } else if (walletState.mode === 'server-managed' && walletState.walletId) {
      await callWalletAPI('clear_wallet', { walletId: walletState.walletId });
    }

    setWalletState({
      mode: null,
      eSpaceAddress: null,
      coreAddress: null,
      walletId: null,
      sessionId: null,
      isLoading: false,
      error: null,
    });
  }, [walletState.mode, walletState.sessionId, walletState.walletId, callWalletAPI]);

  return {
    ...walletState,
    isWalletConnected: !!walletState.walletId,
    isUserWalletConnected: isConnected,
    userAddress: address,
    loadServerWallet,
    delegateUserWallet,
    sendTransaction,
    clearWallet,
  };
}

'use client';

import { useState } from 'react';

export function WalletModeSelector() {

  const {
    mode,
    eSpaceAddress,
    coreAddress,
    isLoading,
    error,
    isWalletConnected,
    isUserWalletConnected,
    userAddress,
    loadServerWallet,
    delegateUserWallet,
    sendTransaction,
    clearWallet,
  } = useUnifiedWallet();

  const handleModeSelection = async (modeType: WalletMode) => {
    setSelectedMode(modeType);
    
    if (modeType === 'server-managed') {
      await loadServerWallet('user123'); // Replace with actual user ID

        alert('Please connect your wallet first');
        return;
      }
      await delegateUserWallet({
        sessionDuration: 60, // 1 hour

    try {

        to: '0x742d35Cc6634C0532925a3b8D4B17dd07E2b9e84',

      alert(`Transaction failed: ${error.message}`);

        <h2 className="text-2xl font-bold mb-4">

  );
}
