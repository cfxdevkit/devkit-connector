💻 Integration Examples

React Integration

// Install the packages npm install @conflux-devkit/ui-primitives @conflux-devkit/state // Use in your React app import { UIProvider, useContracts, useWallets } from '@conflux-devkit/ui-primitives'; function App() { return ( <UIProvider> <Dashboard /> </UIProvider> ); } function Dashboard() { const { contracts, deployContract } = useContracts(); const { wallets, createWallet } = useWallets(); return ( <div> <h1>My Conflux App</h1> <p>Contracts: {contracts.length}</p> <p>Wallets: {wallets.length}</p> </div> ); }

API Server Integration

// Start the orchestrated API server npm run dev:orchestrated // Use the API endpoints const response = await fetch('http://localhost:3001/api/contracts'); const contracts = await response.json(); const walletResponse = await fetch('http://localhost:3001/api/wallets'); const wallets = await walletResponse.json(); const nodeResponse = await fetch('http://localhost:3001/api/node/status'); const nodeStatus = await nodeResponse.json();
